### 2a. Create a CloudFront Function (viewer-request)

CloudFront → Functions → Create function → name `bannister-redirects` → runtime
`cloudfront-js-2.0`. Paste the code below, Save, **Publish**, then attach it to the
distribution's **Default (*) behavior** as a **Viewer request** function association.
Then create an invalidation for `/*`.

```js
function handler(event) {
  var request = event.request;
  var host = request.headers.host.value;
  var uri = request.uri;

  // Legacy WordPress URLs -> current pages (301)
  var map = {
    '/home/': '/',
    '/home': '/',
    '/our-services/': '/services.html',
    '/our-services': '/services.html',
    '/contact-us/': '/contact.html',
    '/contact-us': '/contact.html',
    '/service-area/': '/maryborough.html',
    '/service-area': '/maryborough.html',
    '/cctv-security-cameras/': '/cctv-installation.html',
    '/cctv-security-cameras': '/cctv-installation.html',
    '/about-bannister-communications/': '/about.html',
    '/about-bannister-communications': '/about.html',
    '/remote-support-booking/': '/contact.html',
    '/remote-support-booking': '/contact.html',
    '/computer-maintenance/': '/services.html',
    '/computer-maintenance': '/services.html',
    '/software-purchase-remote-installs/': '/services.html',
    '/software-purchase-remote-installs': '/services.html'
  };

  var qs = Object.keys(request.querystring).length
    ? '?' + Object.keys(request.querystring)
        .map(function (k) { return k + '=' + request.querystring[k].value; })
        .join('&')
    : '';

  // 1. Non-www -> www (301), preserving path + query
  if (host === 'bannistercommunications.com') {
    var target = map[uri] || uri;
    return {
      statusCode: 301,
      statusDescription: 'Moved Permanently',
      headers: { location: { value: 'https://www.bannistercommunications.com' + target + (map[uri] ? '' : qs) } }
    };
  }

  // 2. Legacy path on www -> new page (301)
  if (map[uri]) {
    return {
      statusCode: 301,
      statusDescription: 'Moved Permanently',
      headers: { location: { value: 'https://www.bannistercommunications.com' + map[uri] } }
    };
  }

  return request;
}
```
